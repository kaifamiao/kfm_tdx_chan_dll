{============================================================================}
{ KFM_CHAN 缠论主图指标（DLL 9号槽 kfm_chan_DLL.dll）                          }
{ 内容：K线包含处理 / 分型 / 笔 / 线段 / 笔中枢                                 }
{ 画线方法：主图叠加（建公式时必须勾选）                                        }
{============================================================================}

{—— 第一步：把 HIGH/CLOSE 序列存入 DLL（必须在所有 TDXDLL9(2,...) 之前） ——}
HH:=TDXDLL9(1,HIGH,0,0);
CC:=TDXDLL9(3,CLOSE,0,0);   {CLOSE 供买卖点 MACD 背驰判断}

{—— 第二步：按选择器取各类缠论输出 ——}
FXG:=TDXDLL9(2,LOW,1,0);    {顶分型值}
FXD:=TDXDLL9(2,LOW,2,0);    {底分型值}
ZG:=TDXDLL9(2,LOW,5,0);     {中枢上沿}
ZD:=TDXDLL9(2,LOW,6,0);     {中枢下沿}
MKH:=TDXDLL9(2,LOW,7,0);    {被合并K线组高}
MKL:=TDXDLL9(2,LOW,8,0);    {被合并K线组低}
BIG:=TDXDLL9(2,LOW,9,0);    {笔顶}
BID:=TDXDLL9(2,LOW,10,0);   {笔底}
XDG:=TDXDLL9(2,LOW,11,0);   {线段顶}
XDD:=TDXDLL9(2,LOW,12,0);   {线段底}
ZGS:=TDXDLL9(2,LOW,16,0);   {中枢起点ZG}
ZGE:=TDXDLL9(2,LOW,17,0);   {中枢终点ZG}
ZDS:=TDXDLL9(2,LOW,18,0);   {中枢起点ZD}
ZDE:=TDXDLL9(2,LOW,19,0);   {中枢终点ZD}
B1:=TDXDLL9(2,LOW,20,0);    {一买}
B2:=TDXDLL9(2,LOW,21,0);    {二买}
B3:=TDXDLL9(2,LOW,22,0);    {三买}
S1:=TDXDLL9(2,LOW,23,0);    {一卖}
S2:=TDXDLL9(2,LOW,24,0);    {二卖}
S3:=TDXDLL9(2,LOW,25,0);    {三卖}
PB1:=TDXDLL9(2,LOW,26,0);   {预警一买（分型密封即出，早于确认层）}
PB2:=TDXDLL9(2,LOW,27,0);   {预警二买}
PB3:=TDXDLL9(2,LOW,28,0);   {预警三买}
PS1:=TDXDLL9(2,LOW,29,0);   {预警一卖}
PS2:=TDXDLL9(2,LOW,30,0);   {预警二卖}
PS3:=TDXDLL9(2,LOW,31,0);   {预警三卖}

{⓪ K线：渐变实体样式（底层先画，缠论图层叠加其上）}
PPBBP:=C>0;

STICKLINE(PPBBP AND C>O,C,O,5,6),COLORBLACK;
STICKLINE(PPBBP AND C<O,C,O,5,0),COLORBLACK;

STICKLINE(C>O,C,O,3,0),COLOR000099;
STICKLINE(C>O,C,O,2,0),COLOR0000CC;
STICKLINE(C>O,C,O,1,0),COLOR0000FF;
STICKLINE(C>O,C,O,0,0),COLORCC66FF;{阳实体}

STICKLINE(O>=C,C,O,3,0),COLOR777700;
STICKLINE(O>=C,C,O,2,0),COLOR999900;
STICKLINE(O>=C,C,O,1,0),COLORBBBB00;
STICKLINE(O>=C,C,O,0,0),COLOREEEE00;{阴实体}

{① K线包含处理：被合并的K线组画灰色空心框}
STICKLINE(MKH>0,MKH,MKL,1,1),COLORLIGRAY;

{② 分型标注：顶/底}
DRAWTEXT(FXG>0,FXG*1.002,'顶'),COLORRED;
DRAWTEXT(FXD>0,FXD*0.998,'底'),COLORGREEN;

{③ 笔：黄色折线连接 + 端点价格数值}
DRAWLINE(BIG>0,BIG,BID>0,BID,0),COLORYELLOW;
DRAWLINE(BID>0,BID,BIG>0,BIG,0),COLORYELLOW;
DRAWNUMBER(BIG>0,BIG*1.004,BIG),COLORRED;
DRAWNUMBER(BID>0,BID*0.996,BID),COLORGREEN;

{④ 线段：品红粗折线 + 端点名称}
DRAWLINE(XDG>0,XDG,XDD>0,XDD,0),COLORMAGENTA,LINETHICK2;
DRAWLINE(XDD>0,XDD,XDG>0,XDG,0),COLORMAGENTA,LINETHICK2;
DRAWTEXT(XDG>0,XDG*1.008,'段顶'),COLORMAGENTA;
DRAWTEXT(XDD>0,XDD*0.992,'段底'),COLORLIBLUE;

{⑤ 中枢：第三版——纯STICKLINE四边矩形（DRAWLINE会把所有触发点连成折线，
   产生跨中枢连线，故中枢不用DRAWLINE；上下沿为逐bar细柱，左右边用独立标记）}
STICKLINE(ZG>0,ZG,ZG*1.003,1,0),COLORYELLOW;      {上沿}
STICKLINE(ZG>0,ZD,ZD*0.997,1,0),COLORYELLOW;      {下沿}
STICKLINE(ZGS>0,ZGS,ZDS,0,0),COLORYELLOW;         {左边}
STICKLINE(ZGE>0,ZGE,ZDE,0,0),COLORYELLOW;         {右边}
DRAWTEXT(ZGS>0,ZGS*1.006,'ZG'),COLORYELLOW;
DRAWNUMBER(ZGS>0,ZGS,ZGS),COLORYELLOW;
DRAWTEXT(ZGS>0,ZDS*0.994,'ZD'),COLORYELLOW;
DRAWNUMBER(ZGS>0,ZDS,ZDS),COLORYELLOW;

{⑥ 买卖点标注（笔级别，MACD背驰+中枢位置；只标已确认端点，永不漂移）}
DRAWTEXT(B1>0,B1*0.985,'1买'),COLORRED;
DRAWTEXT(B2>0,B2*0.985,'2买'),COLORLIRED;
DRAWTEXT(B3>0,B3*0.985,'3买'),COLORMAGENTA;
DRAWTEXT(S1>0,S1*1.015,'1卖'),COLORGREEN;
DRAWTEXT(S2>0,S2*1.015,'2卖'),COLORLIGREEN;
DRAWTEXT(S3>0,S3*1.015,'3卖'),COLORLIBLUE;

{⑧ 预警信号：分型密封即出（比确认层早约2-4根K线），密封后固定不漂移}
DRAWTEXT(PB1>0,PB1*0.978,'预1买'),COLORCYAN;
DRAWTEXT(PB2>0,PB2*0.978,'预2买'),COLORCYAN;
DRAWTEXT(PB3>0,PB3*0.978,'预3买'),COLORCYAN;
DRAWTEXT(PS1>0,PS1*1.022,'预1卖'),COLORWHITE;
DRAWTEXT(PS2>0,PS2*1.022,'预2卖'),COLORWHITE;
DRAWTEXT(PS3>0,PS3*1.022,'预3卖'),COLORWHITE;

{⑦ 实时锚定备选：第4参数传1，信号标在"确认bar"（信号出现时的新K线）上，
   完全不补历史位置。想用就删掉上面六行、启用下面六行}
{B1R:=TDXDLL9(2,LOW,20,1);}
{B2R:=TDXDLL9(2,LOW,21,1);}
{B3R:=TDXDLL9(2,LOW,22,1);}
{S1R:=TDXDLL9(2,LOW,23,1);}
{S2R:=TDXDLL9(2,LOW,24,1);}
{S3R:=TDXDLL9(2,LOW,25,1);}
{DRAWICON(B1R>0 OR B2R>0 OR B3R>0,LOW,1);}
{DRAWICON(S1R>0 OR S2R>0 OR S3R>0,HIGH,2);}

{备选：整面空心柱带（第一版风格），删掉上面八行、启用下面一行}
{STICKLINE(ZG>0,ZG,ZD,1,1),COLORYELLOW;}
